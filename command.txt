# arookieofcDB Command Test Script
# 用法：逐行执行，便于观察每步输出与 status 指标。

# 0) 基础
help
status

# 1) 引擎切换与基础读写
engine memory
set k1 v1
get k1
status

engine wal
set k1 v1
get k1
status

engine hybrid
set k1 v1
get k1
status

# 2) select --disk（重点）
# 在 hybrid 下：get 走缓存优先，select --disk 强制走持久层且不回填缓存
engine hybrid
set sel:1 hello
get sel:1
select sel:1
select --disk sel:1
status

# 3) cachemax（命令设置缓存上限）
# 0 表示不限制
engine hybrid
cachemax
cachemax 1
set c1 v1
set c2 v2
status
# 观察: cache_evictions 应增加，cache_current_keys 应为 1
get c1
status
# 观察: c1 会从磁盘回源
cachemax 0
status

# 4) TTL 持久化验证（WAL）
engine wal
set ttl:persist alive
expire ttl:persist 30
ttl ttl:persist
status
# 现在退出并重启程序后执行：
# engine wal
# get ttl:persist
# ttl ttl:persist

# 5) 停机期间过期验证（WAL）
engine wal
set ttl:down will_expire
expire ttl:down 1
status
# 现在退出，等待 2 秒，再重启执行：
# engine wal
# get ttl:down
# ttl ttl:down

# 6) SQL-like
engine hybrid
add user:1 alice
select user:1
update user:1 bob
select user:1
delete user:1
select user:1
status

# 7) 数值命令
engine hybrid
set count 100
incr count
incrby count 10
set ratio 3.14
incrbyfloat ratio 0.86
get count
get ratio
status

# 8) 结束
exit

# 9) cachepolicy（缓存策略）
engine hybrid
cachepolicy
cachepolicy none
set pn 1
get pn
status
# 观察: cache_policy=none 且 cache_current_keys 不增长
cachepolicy lru
get pn
status
# 观察: 切回 lru 后读取会回填缓存

# 10) 一致性验证与修复（hybrid）
engine hybrid
set vr:key disk
verify
# 如需修复（以磁盘为准 / 以缓存为准）：
repair --to disk
verify
repair --to cache
verify

# 11) status 一致性健康度
engine hybrid
status
# 关注: inconsistency_total / inconsistency_only_in_cache / inconsistency_only_in_disk / inconsistency_value_mismatch
verify
repair --to cache
status

# 12) 最近修复摘要（last_repair_*）
engine hybrid
verify
status
repair --to cache
status
# 关注: last_repair_target/last_repair_total/last_repair_only_in_cache/last_repair_only_in_disk/last_repair_value_mismatch

# 13) 故障注入演示（fault）
engine hybrid

fault cache-only demo:key cache_v
verify
status
repair --to disk
verify
status

set demo2:key base
fault disk-only demo2:key disk_v
verify

verify
status

# 14) repairmode 自动修复模式
engine hybrid
repairmode
repairmode read
fault cache-only auto:key stale
get auto:key
verify
status
repairmode off
status

# 15) hybrid: update/delete 后 get 与 disk 一致（不读旧值）
engine hybrid
set hy:order:1 v1
get hy:order:1
select --disk hy:order:1
set hy:order:1 v2
get hy:order:1
select --disk hy:order:1
del hy:order:1
get hy:order:1
select --disk hy:order:1
# 预期: get 与 select --disk 全程一致，删除后都为 (nil)

# 16) hybrid: cache miss 回源并回填一致
engine hybrid
cachepolicy none
set hy:miss:1 from-disk
status
# 预期: cache_current_keys=0
cachepolicy lru
get hy:miss:1
status
select --disk hy:miss:1
# 预期: get 与 disk 一致，且回填后 cache_current_keys=1
